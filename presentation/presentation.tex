% Endzeit auf Notizfolien einfügen
\newcounter{accumsec} % akkumulierte Zeit in Sekunden
\setcounter{accumsec}{0}
% Zeit hinzufügen (in Sekunden)
\newcommand{\addseconds}[1]{%
  \addtocounter{accumsec}{#1}%
}
% Sekunden in mm:ss umwandeln
\newcommand{\secstommss}[1]{%
  \begingroup
  \count0=#1\relax
  \count1=\count0
  \divide\count1 by 60 % Minuten
  \count2=\count0
  \advance\count2 by -\numexpr\count1*60\relax % Sekunden
  \ifnum\count2<10  % führende Null
    \edef\result{\the\count1:0\the\count2}%
  \else
    \edef\result{\the\count1:\the\count2}%
  \fi
  \result
  \endgroup
}
% zeigt Zeit für Overlay und akkumulierte Gesamtzeit auf Notizseiten an 
\newcommand{\notetime}{}
\newcommand{\setnotetime}[1]{\renewcommand{\notetime}{#1}}
\setbeamertemplate{note page}{
  \vbox to \textheight{
    \insertnote
    \vfill
    \hbox to \linewidth{
      \hfill
      \color{red}{%
        \ifx\notetime\@empty
        \else
			\Large\notetime\;/\;\secstommss{\value{accumsec}}        
		\fi
      }
    }
  }
}
% bekommt Sekunden für spezielle Overlay
\newcommand<>{\timednote}[2]{%
  \note#3{%
    \setnotetime{{\secstommss{#1}}}%
    #2%
	\addseconds{#1}%
  }%
}
